// ======================================================
// 二宮金次郎 (銅像敵キャラ) ベクタークラス - NinomiyaKinjiroEnemy.js
// カクカク動作 & ローリングアタック対応
// ======================================================

class NinomiyaKinjiroEnemy {
    constructor(x, y, config = {}) {
        this.x = x;
        this.y = y;
        this.vx = 0;
        this.vy = 0;
        this.size = config.size || 60; // サイズ(基準半径)
        
        // ステート管理: IDLE, WALK, CHARGE, ROLLING, DAZED, HURT
        this.state = 'IDLE'; 
        
        // カクカク度の設定 (描画コマ数制限)
        this.choppyFps = config.choppyFps || 6; 
        this.choppyTimer = 0;
        this.renderPoseFrame = 0;
        
        // ローリングアタック物理パラメータ
        this.rollAngle = 0;
        this.rollSpeed = config.rollSpeed || 18;
        this.chargeTimer = 0;
        this.rollTimer = 0;
        
        // マテリアルカラー (bronze, patina, gold)
        this.material = config.material || 'bronze';
        
        // AI設定
        this.target = null;
        this.attackCooldown = 0;
    }

    // マテリアルカラーパレット取得
    getColors() {
        switch(this.material) {
            case 'patina': // 錆びた緑青
                return { main: '#2a5a4b', dark: '#16382e', light: '#458572', highlight: '#73b3a0', book: '#884422' };
            case 'gold': // 黄金二宮
                return { main: '#d97706', dark: '#78350f', light: '#fbbf24', highlight: '#fef08a', book: '#451a03' };
            case 'bronze': // クラシックブロンズ
            default:
                return { main: '#785e3a', dark: '#4d3b23', light: '#a38352', highlight: '#cbb087', book: '#5c1d11' };
        }
    }

    // 状態のアップデート (毎フレーム呼出)
    update(dt, targetPos) {
        this.attackCooldown -= dt;
        
        // カクカクコマ送りアニメーションのステップ更新
        this.choppyTimer += dt;
        if (this.choppyTimer >= 1 / this.choppyFps) {
            this.choppyTimer = 0;
            this.renderPoseFrame++;
        }

        // AIロジック
        switch (this.state) {
            case 'IDLE':
                this.vx = 0;
                this.vy = 0;
                if (targetPos) {
                    const dist = Math.hypot(targetPos.x - this.x, targetPos.y - this.y);
                    if (dist < 300) this.state = 'WALK';
                }
                break;

            case 'WALK':
                if (targetPos) {
                    const dx = targetPos.x - this.x;
                    const dy = targetPos.y - this.y;
                    const angle = Math.atan2(dy, dx);
                    // カクカク移動 (一定間隔でカクッと進む)
                    const speed = 2.5;
                    this.vx = Math.cos(angle) * speed;
                    this.vy = Math.sin(angle) * speed;

                    const dist = Math.hypot(dx, dy);
                    if (dist < 200 && this.attackCooldown <= 0) {
                        this.startRollingAttack(angle);
                    }
                }
                break;

            case 'CHARGE': // 溜め（本を閉じ、目が赤く発光）
                this.vx *= 0.8;
                this.vy *= 0.8;
                this.chargeTimer -= dt;
                if (this.chargeTimer <= 0) {
                    this.state = 'ROLLING';
                    this.rollTimer = 1.5; // 1.5秒間回転攻撃
                }
                break;

            case 'ROLLING': // 高速ローリングアタック！
                this.rollAngle += this.rollSpeed * dt * 10;
                this.x += this.vx;
                this.y += this.vy;
                this.rollTimer -= dt;
                if (this.rollTimer <= 0) {
                    this.state = 'DAZED'; // 攻撃後の隙
                    this.dazedTimer = 1.2;
                    this.attackCooldown = 3.0; // クールダウン
                }
                break;

            case 'DAZED':
                this.vx = 0;
                this.vy = 0;
                this.dazedTimer -= dt;
                if (this.dazedTimer <= 0) this.state = 'IDLE';
                break;
        }

        this.x += this.vx;
        this.y += this.vy;
    }

    startRollingAttack(angle) {
        this.state = 'CHARGE';
        this.chargeTimer = 0.6; // 0.6秒の溜め
        this.targetAngle = angle;
        // 攻撃方向に加速ベクトルをセット
        this.vx = Math.cos(angle) * 12;
        this.vy = Math.sin(angle) * 12;
    }

    // ベクター描画関数
    draw(ctx) {
        ctx.save();
        ctx.translate(this.x, this.y);

        const c = this.getColors();
        const f = this.renderPoseFrame; // カクカク動かすためのフレーム数

        if (this.state === 'ROLLING') {
            // --- ローリングアタック描画 (高速回転) ---
            ctx.rotate(this.rollAngle);
            
            // 残像・回転エフェクト
            ctx.beginPath();
            ctx.arc(0, 0, this.size * 0.8, 0, Math.PI * 2);
            ctx.fillStyle = c.dark;
            ctx.fill();

            // 薪がぐるぐる回るシルエット
            for (let i = 0; i < 4; i++) {
                ctx.rotate(Math.PI / 2);
                ctx.fillStyle = c.main;
                ctx.fillRect(-this.size * 0.6, -10, this.size * 1.2, 20);
                // 薪の断面
                ctx.fillStyle = c.highlight;
                ctx.beginPath();
                ctx.arc(this.size * 0.6, 0, 8, 0, Math.PI * 2);
                ctx.fill();
            }

            // 回転の中心で光る本と赤目
            ctx.fillStyle = c.book;
            ctx.fillRect(-12, -10, 24, 20);
            ctx.fillStyle = '#ff0000';
            ctx.beginPath();
            ctx.arc(-4, -2, 3, 0, Math.PI * 2);
            ctx.arc(4, -2, 3, 0, Math.PI * 2);
            ctx.fill();

        } else {
            // --- 立ち/歩行/溜め 姿勢描画 (カクカクモーション) ---
            
            // カクカクした上下・揺れオフセット
            const stepOffset = (this.state === 'WALK') ? ((f % 2 === 0) ? -4 : 4) : 0;
            const legAngle = (this.state === 'WALK') ? ((f % 2 === 0) ? 0.2 : -0.2) : 0;

            // 台座 (移動するときは地面をこする)
            ctx.fillStyle = '#334155';
            ctx.fillRect(-25, 25, 50, 10);

            // 1. 背中の薪 (Oikura / Firewood)
            ctx.save();
            ctx.translate(-10, -5 + stepOffset);
            ctx.fillStyle = c.dark;
            // 木枠
            ctx.fillRect(-8, -35, 6, 40);
            // 薪の束 (ベクターの束)
            for(let i=0; i<5; i++) {
                ctx.fillStyle = (i % 2 === 0) ? c.main : c.light;
                ctx.fillRect(-15, -35 + (i * 7), 22, 5);
            }
            ctx.restore();

            // 2. 脚 (カクカク関節)
            ctx.save();
            ctx.translate(0, 10);
            ctx.fillStyle = c.dark;
            // 左脚
            ctx.fillRect(-12, 0, 8, 15);
            // 右脚 (歩行時回転)
            ctx.rotate(legAngle);
            ctx.fillRect(4, 0, 8, 15);
            ctx.restore();

            // 3. 胴体 (着物・ブロンズ)
            ctx.fillStyle = c.main;
            ctx.beginPath();
            ctx.moveTo(-16, 12 + stepOffset);
            ctx.lineTo(-12, -20 + stepOffset);
            ctx.lineTo(12, -20 + stepOffset);
            ctx.lineTo(16, 12 + stepOffset);
            ctx.closePath();
            ctx.fill();

            // 帯
            ctx.fillStyle = c.dark;
            ctx.fillRect(-14, -4 + stepOffset, 28, 6);

            // 4. 頭部 & チョンマゲ
            ctx.save();
            ctx.translate(0, -28 + stepOffset);
            
            // 溜め時は少しうつむく
            if (this.state === 'CHARGE') ctx.rotate(0.2);

            ctx.fillStyle = c.light;
            ctx.beginPath();
            ctx.arc(0, 0, 12, 0, Math.PI * 2);
            ctx.fill();

            // 目 (敵役なので鋭い＆発光)
            if (this.state === 'CHARGE') {
                ctx.fillStyle = '#ef4444'; // 攻撃溜め時は真っ赤
                ctx.shadowColor = '#ef4444';
                ctx.shadowBlur = 10;
            } else {
                ctx.fillStyle = '#000000';
            }
            ctx.fillRect(2, -2, 4, 3); // 前を睨む目

            ctx.restore();

            // 5. 腕 & 本 (二宮金次郎の象徴)
            ctx.save();
            ctx.translate(6, -12 + stepOffset);
            
            // 本の保持手
            ctx.fillStyle = c.main;
            ctx.fillRect(0, -2, 12, 6); // 腕

            // 本 (敵モード時、溜めでガバッと開く)
            ctx.translate(10, -2);
            if (this.state === 'CHARGE') ctx.rotate(-0.4);

            ctx.fillStyle = c.book;
            ctx.fillRect(0, -8, 10, 14);
            ctx.fillStyle = '#f8fafc'; // ページ
            ctx.fillRect(2, -7, 7, 12);

            ctx.restore();
        }

        ctx.restore();
    }
}
